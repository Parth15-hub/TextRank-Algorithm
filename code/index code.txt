        <div>
          <div class="step-title">Text Preprocessing (Shared Pipeline)</div>
          <div class="step-desc">All models share the same preprocessing: tokenization, stopword removal, lowercasing, and punctuation cleaning to reduce noise before analysis.</div>

          <div class="step-desc">Scores each sentence by summing all TF-IDF term weights. The highest-scoring sentences are selected directly — no graph construction needed. Fast, simple, interpretable.</div>
          <div class="step-code">score(s) = Σ TF(t,s) × IDF(t,D)  →  top-N by score</div>
        </div>
      </div>
      <div class="step-item">
        <span class="step-num">03</span>
        <div>
          <div class="step-title">TextRank (Graph + PageRank)</div>
          <div class="step-desc">Builds a sentence similarity graph using cosine similarity. Applies the PageRank-inspired TextRank algorithm to iteratively compute sentence importance until convergence.</div>
          <div class="step-code">WS(Vᵢ) = (1-d) + d × Σ [ w_ji / Σ w_jk × WS(Vⱼ) ]</div>
        </div>

        <div>

async function uploadAndExtract(file) {
  if (!file) return;

  const allowed = [".txt", ".pdf", ".docx"];
  const ext = file.name.slice(file.name.lastIndexOf(".")).toLowerCase();
  if (!allowed.includes(ext)) {
    setUploadStatus("err", `Unsupported format: ${ext}. Please use .txt, .pdf, or .docx`);
    return;
  }

  setUploadStatus("loading", `Extracting text from "${file.name}"…`);

  const formData = new FormData();
  formData.append("article", file);

  try {
    const resp = await fetch("http://localhost:3000/api/upload", {
      method: "POST",
      body: formData
    });

    const data = await resp.json().catch(() => ({}));

    if (!resp.ok) {
      setUploadStatus("err", data.error || `Upload failed (HTTP ${resp.status})`);
      return;
    }

    // Populate textarea with extracted text
    const ta = document.getElementById("inputText");
    ta.value = data.text || "";
    document.getElementById("wordCount").textContent = `${data.wordCount || 0} words`;

    setUploadStatus("ok", `✓ Loaded "${file.name}" — ${data.wordCount} words extracted`);

    // Auto-scroll to show the populated textarea
    ta.scrollIntoView({ behavior: "smooth", block: "nearest" });

  } catch (err) {
    setUploadStatus("err", `Upload error: ${err.message}. Make sure the server is running.`);
  }
}

// BERT via HuggingFace Inference API
// ─────────────────────────────────────────────────────────────────────────────
async function summarizeWithBERT(text, numSentences) {
  const key = document.getElementById("hfApiKey").value.trim();
  // Truncate to ~1000 words for API limits
  const truncated = text.split(/\s+/).slice(0, 800).join(" ");

  const headers = { "Content-Type": "application/json" };
  if (key) headers["Authorization"] = `Bearer ${key}`;

  const maxLen = Math.min(150, numSentences * 50);
  const minLen = Math.max(30, numSentences * 20);

  const resp = await fetch(
    "https://api-inference.huggingface.co/models/facebook/bart-large-cnn",
    {
      method: "POST",
      headers,
      body: JSON.stringify({
        inputs: truncated,
        parameters: { max_length: maxLen, min_length: minLen, do_sample: false }
      })
    }
  );

  if (!resp.ok) {
    const err = await resp.json().catch(() => ({}));
    if (resp.status === 503) throw new Error("HuggingFace model is loading. Try again in ~20 seconds.");
    if (resp.status === 401) throw new Error("HuggingFace API key invalid or required. Get a free key at huggingface.co");
    throw new Error(err.error || `HuggingFace API error (${resp.status})`);
  }

  const data = await resp.json();
  const summary = data?.[0]?.summary_text || data?.summary_text || "";
  if (!summary) throw new Error("BERT returned an empty summary.");
  return summary;
}

async function runSummarize() {
  const text = document.getElementById("inputText").value.trim();
  if (!text || text.length < 50) { showError("Please enter a longer article."); return; }

  hideError();
  setLoading(true);

  const numSentences = Number(document.getElementById("summaryLen").value);
  const refSummary = document.getElementById("refSummary").value.trim();
  const loaderMsg = MODEL_META[currentModel]?.loader || "Processing...";
  document.getElementById("loaderText").textContent = loaderMsg;

  try {
    let summary, rouge = null, graphData = null;

    if (currentModel === "bert") {
      summary = await summarizeWithBERT(text, numSentences);
      // BERT is abstractive — no graph data
    } else {
      const response = await fetch("http://localhost:3000/api/summarize", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ text, numSentences, model: currentModel, referenceSummary: refSummary || undefined })
      });

      const data = await response.json().catch(() => ({}));
      if (!response.ok) throw new Error(data?.error || `HTTP ${response.status}`);

      summary = data.summary || "";
      rouge = data.rouge || null;
      graphData = data.graphData || null;

      if (!summary) throw new Error("No summary returned");
    }

    lastSummary = summary;
    lastGraphData = graphData;

    // Stats
    const inputWords = text.split(/\s+/).filter(Boolean).length;
    const outputWords = summary.split(/\s+/).filter(Boolean).length;
    const reduction = inputWords ? Math.round((1 - outputWords / inputWords) * 100) : 0;

    document.getElementById("metaSentences").textContent = String(numSentences);
    document.getElementById("metaReduction").textContent = `${reduction}%`;
    document.getElementById("metaWords").textContent = String(outputWords);
    document.getElementById("metaModel").textContent = currentModel.toUpperCase();

    const safe = summary.replace(/[&<>"']/g, ch => ({"&":"&amp;","<":"&lt;",">":"&gt;",'"':"&quot;","'":"&#39;"}[ch]));
    document.getElementById("outputText").innerHTML = safe;
    document.getElementById("outStatus").textContent = "complete";
    document.getElementById("outputMeta").style.display = "flex";

    // ROUGE display
    if (rouge) showRouge(rouge);

    // Graph
    if (graphData) renderD3Graph(graphData);

  } catch (err) {
    showError(String(err).slice(0, 500));
    console.error(err);
  } finally {
    setLoading(false);
  }
}

  // ── Zoom behaviour ───────────────────────────────────────────────────────
  // All graph content lives inside a <g id="zoomGroup"> so zoom/pan only
  // transforms that group — the SVG boundary never changes.
  d3Zoom = d3.zoom()
    .scaleExtent([0.3, 4])
    .on("zoom", (event) => {
      d3Svg.select("#zoomGroup").attr("transform", event.transform);
    });

  d3Svg.call(d3Zoom);

  const g = d3Svg.append("g").attr("id", "zoomGroup");

  // Arrow defs
  d3Svg.append("defs").append("marker")
    .attr("id", "arrow")
    .attr("viewBox", "0 -5 10 10")
    .attr("refX", 15).attr("refY", 0)
    .attr("markerWidth", 4).attr("markerHeight", 4)
    .attr("orient", "auto")
    .append("path").attr("d", "M0,-5L10,0L0,5")
    .attr("fill", "rgba(200,240,96,0.4)");

  // ── Simulation ───────────────────────────────────────────────────────────
  // forceCenter keeps the cluster centred.
  // forceCollide prevents node overlap.
  // forceBoundingBox (custom) clamps positions inside [PAD, W-PAD] × [PAD, H-PAD].
  const nodeRadius = d => 12 + (d.normScore || 0) * 12;

  d3SimRef = d3.forceSimulation(nodes)
    .force("link",      d3.forceLink(edges)
                          .id(d => d.id)
                          .distance(d => Math.max(60, 130 - (d.weight || 0) * 60))
                          .strength(d => Math.min(1, (d.weight || 0) * 0.9)))
    .force("charge",    d3.forceManyBody().strength(-200).distanceMax(300))
    .force("center",    d3.forceCenter(W / 2, H / 2).strength(0.08))
    .force("collision", d3.forceCollide().radius(d => nodeRadius(d) + 10).strength(0.9))
    .alphaDecay(0.025)
    .velocityDecay(0.4);

  // Bounding-box force: applied on every tick to clamp x/y
  d3SimRef.force("bounds", () => {
    for (const node of nodes) {
      const r = nodeRadius(node) + 4;
      node.x = Math.max(PAD + r, Math.min(W - PAD - r, node.x || W / 2));
      node.y = Math.max(PAD + r, Math.min(H - PAD - r, node.y || H / 2));
      // Also clamp velocity so nodes can't tunnel through the wall on fast frames
      if (Math.abs(node.vx) > 8) node.vx *= 0.5;
      if (Math.abs(node.vy) > 8) node.vy *= 0.5;
    }
  });

  // Warm up briefly so nodes start in a reasonable layout, not all at origin
  d3SimRef.tick(80);

  // ── Edges ────────────────────────────────────────────────────────────────
  const link = g.append("g").selectAll("line")
    .data(edges)
    .join("line")
    .attr("stroke",       d => `rgba(200,240,96,${Math.max(0.08, (d.weight || 0) * 0.55)})`)
    .attr("stroke-width", d => Math.max(0.5, (d.weight || 0) * 2.5));

  // ── Node groups ──────────────────────────────────────────────────────────
  const node = g.append("g").selectAll("g")
    .data(nodes)
    .join("g")
    .call(d3.drag()
      .on("start", (e, d) => {
        if (!e.active) d3SimRef.alphaTarget(0.25).restart();
        d.fx = d.x; d.fy = d.y;
      })
      .on("drag", (e, d) => {
        // Clamp drag position to bounds so the user can't drag nodes off-screen
        const r = nodeRadius(d) + 4;
        d.fx = Math.max(PAD + r, Math.min(W - PAD - r, e.x));
        d.fy = Math.max(PAD + r, Math.min(H - PAD - r, e.y));
      })
      .on("end", (e, d) => {
        if (!e.active) d3SimRef.alphaTarget(0);
        d.fx = null; d.fy = null;
      })
    );

  // Glow halo
  node.append("circle")
    .attr("r",    d => nodeRadius(d) * 1.9)
    .attr("fill", d => d.selected ? "rgba(200,240,96,0.09)" : "rgba(74,122,32,0.05)");

  // Main fill circle
  node.append("circle")
    .attr("r",    d => nodeRadius(d))
    .attr("fill", d => {
      const ns = d.normScore || 0;
      return d.selected
        ? `rgb(${Math.round(74 + ns * 126)},${Math.round(74 + ns * 166)},${Math.round(32 + ns * 64)})`
        : "rgb(40,60,20)";
    })
    .attr("stroke",       d => d.selected ? "rgba(200,240,96,0.85)" : "rgba(200,240,96,0.2)")
    .attr("stroke-width", d => d.selected ? 1.5 : 0.5);

  // Label
  node.append("text")
    .text(d => d.label)
    .attr("text-anchor",      "middle")
    .attr("dominant-baseline","middle")
    .attr("font-family",      "'DM Mono', monospace")
    .attr("font-size",        "9px")
    .attr("font-weight",      "bold")
    .attr("fill",             d => d.selected ? "#0a0a08" : "#c8f060")
    .attr("pointer-events",   "none");

  // Tooltip
  const tooltip = document.getElementById("graphTooltip");
  node
    .on("mouseover", (e, d) => {
      tooltip.style.display = "block";
      tooltip.style.left    = (e.offsetX + 14) + "px";
      tooltip.style.top     = (e.offsetY -  8) + "px";
      tooltip.innerHTML     = `<strong>${d.label}</strong>${d.selected ? " ✓" : ""}<br>Score: ${(d.score || 0).toFixed(4)}<br><em style="color:var(--muted)">${d.preview || ""}</em>`;
    })
    .on("mousemove", e => {
      tooltip.style.left = (e.offsetX + 14) + "px";
      tooltip.style.top  = (e.offsetY -  8) + "px";
    })
    .on("mouseout", () => { tooltip.style.display = "none"; });

  // ── Tick handler ─────────────────────────────────────────────────────────
  d3SimRef.on("tick", () => {
    link
      .attr("x1", d => d.source.x)
      .attr("y1", d => d.source.y)
      .attr("x2", d => d.target.x)
      .attr("y2", d => d.target.y);

    node.attr("transform", d => `translate(${d.x},${d.y})`);
  });

  // ── Zoom control buttons ─────────────────────────────────────────────────
  document.getElementById("zoomIn").onclick = () =>
    d3Svg.transition().duration(250).call(d3Zoom.scaleBy, 1.4);

  document.getElementById("zoomOut").onclick = () =>
    d3Svg.transition().duration(250).call(d3Zoom.scaleBy, 0.72);

  document.getElementById("zoomReset").onclick = () =>
    d3Svg.transition().duration(350).call(d3Zoom.transform, d3.zoomIdentity);

  // ── Resize handler ───────────────────────────────────────────────────────
  // Debounced — only recalculates if the container width actually changed
  let prevW = W;
  const resizeObserver = new ResizeObserver(() => {
    const newW = Math.max(300, container.clientWidth - 56);
    if (Math.abs(newW - prevW) < 10) return;
    prevW = newW;

    svgEl.setAttribute("width",   newW);
    svgEl.setAttribute("viewBox", `0 0 ${newW} ${H}`);

    d3SimRef
      .force("center", d3.forceCenter(newW / 2, H / 2).strength(0.08))
      .force("bounds", () => {
        for (const nd of nodes) {
          const r = nodeRadius(nd) + 4;
          nd.x = Math.max(PAD + r, Math.min(newW - PAD - r, nd.x));
          nd.y = Math.max(PAD + r, Math.min(H    - PAD - r, nd.y));
        }
      })
      .alpha(0.3)
      .restart();
  });
  resizeObserver.observe(container);

  document.getElementById("graph-section").scrollIntoView({ behavior: "smooth" });
}

// ─────────────────────────────────────────────────────────────────────────────
// DECORATIVE CANVAS GRAPH (How It Works section — existing animation)
// ─────────────────────────────────────────────────────────────────────────────
const canvas = document.getElementById("graphCanvas");
const ctx = canvas.getContext("2d");
const W = 340, H = 320;
canvas.width = W; canvas.height = H;

const demoNodes = Array.from({ length: 8 }, (_, i) => ({
  id: i, x: 60 + Math.random() * 220, y: 50 + Math.random() * 220,
  vx: (Math.random() - 0.5) * 0.4, vy: (Math.random() - 0.5) * 0.4,
  rank: Math.random(), label: `S${i + 1}`,
}));
demoNodes[0].rank = 0.95; demoNodes[2].rank = 0.78; demoNodes[5].rank = 0.62;

const demoEdges = [];
for (let i = 0; i < demoNodes.length; i++)
  for (let j = i + 1; j < demoNodes.length; j++)
    if (Math.random() > 0.45) demoEdges.push({ a: i, b: j, w: 0.2 + Math.random() * 0.8 });

function lerp(a, b, t) { return a + (b - a) * t; }

function drawGraph(t) {
  ctx.clearRect(0, 0, W, H);
  demoNodes.forEach(n => {
    n.x += n.vx; n.y += n.vy;
    if (n.x < 30 || n.x > W - 30) n.vx *= -1;
    if (n.y < 30 || n.y > H - 30) n.vy *= -1;
    demoNodes.forEach(m => {
      if (m === n) return;
      const dx = n.x - m.x, dy = n.y - m.y;
      const d = Math.sqrt(dx*dx + dy*dy);
      if (d < 80) { n.vx += (dx/d)*0.05; n.vy += (dy/d)*0.05; }
    });
    n.vx *= 0.98; n.vy *= 0.98;
  });
  demoEdges.forEach(e => {
    const a = demoNodes[e.a], b = demoNodes[e.b];
    const pulse = 0.5 + 0.5 * Math.sin(t * 0.002 + e.a + e.b);
    ctx.beginPath();
    ctx.moveTo(a.x, a.y); ctx.lineTo(b.x, b.y);
    ctx.strokeStyle = `rgba(200, 240, 96, ${e.w * 0.3 + pulse * 0.15})`;
    ctx.lineWidth = e.w * 1.5; ctx.stroke();
  });
  demoNodes.forEach(n => {
    const r = 8 + n.rank * 12;
    const pulse = 1 + 0.08 * Math.sin(t * 0.003 + n.id);
    const grd = ctx.createRadialGradient(n.x, n.y, 0, n.x, n.y, r * 2.5);
    grd.addColorStop(0, `rgba(200,240,96,${n.rank * 0.3})`);
    grd.addColorStop(1, "rgba(200,240,96,0)");
    ctx.beginPath(); ctx.arc(n.x, n.y, r * 2.5 * pulse, 0, Math.PI * 2);
    ctx.fillStyle = grd; ctx.fill();
    ctx.beginPath(); ctx.arc(n.x, n.y, r * pulse, 0, Math.PI * 2);
    const g = Math.round(lerp(74, 240, n.rank));
    ctx.fillStyle = `rgb(${Math.round(lerp(74, 200, n.rank))}, ${g}, ${Math.round(lerp(32, 96, n.rank))})`;
    ctx.fill(); ctx.strokeStyle = "rgba(200,240,96,0.4)"; ctx.lineWidth = 1; ctx.stroke();
    ctx.fillStyle = n.rank > 0.6 ? "#0a0a08" : "#c8f060";
    ctx.font = "bold 9px 'DM Mono', monospace";
    ctx.textAlign = "center"; ctx.textBaseline = "middle";
    ctx.fillText(n.label, n.x, n.y);
  });
}
let frame = 0;
(function animate() { drawGraph(frame++); requestAnimationFrame(animate); })();
